### Standalone function

The standalone function version of this method:

```typescript
import { AgentmailCliCore } from "agentmail/core.js";
import { metricsMetricsQuery } from "agentmail/funcs/metrics-metrics-query.js";

// Use `AgentmailCliCore` for best tree-shaking performance.
// You can create one instance of it to use across an application.
const agentmailCli = new AgentmailCliCore({
  bearerAuth: "<YOUR_BEARER_TOKEN_HERE>",
});

async function run() {
  const res = await metricsMetricsQuery(agentmailCli, {});
  if (res.ok) {
    const { value: result } = res;
    console.log(result);
  } else {
    console.log("metricsMetricsQuery failed:", res.error);
  }
}

run();
```
